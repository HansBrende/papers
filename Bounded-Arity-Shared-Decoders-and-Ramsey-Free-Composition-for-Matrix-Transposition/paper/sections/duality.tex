\section{Anti-diagonal untwisting and operator entanglement duality}\label{sec:duality}

\subsection{Pairwise shift collisions, anti-diagonal tensor separation, and mask floors}\label{sec:antidiag}

Problem~9.1 of \cite{SeventeenthPaper2026} (together with the characteristic-split constructions
and blow-ups of \cite{ClassUniformMemo2026}) asks what structural constraints govern shared decoders
on aperiodic class partitions $p : \Z_d \to \kappa$, while \cite[Theorem~3.4 and
Appendix~C]{SharedDecoders2026} proved the universal capacity bound
$\dim_F C \le \frac{41}{64} b d^2 + \frac{5}{3} b d^{3/2}$ ($M = 2$) alongside probabilistic-method
floors showing that random-mask certificate arguments stall at $9/16$ (full-ambient isolation) and
$5/8$ (nested component counting).

Our first observation is that \emph{every} shift-collision condition on a linear source
$C \le \Mat_{d\times d}(A)$ decouples into $d$ completely independent $1$-dimensional slices once
we untwist matrix coordinates along anti-diagonals $a = c + j \in \Z_d$. Specifically, define the
mutually inverse $F$-linear maps (\lean{antiDiagLinear}, \lean{fromAntiDiagLinear})
\[
  Z_{c,a} \;:=\; X_{c,\,a - c},
  \qquad
  X_{c,j} \;=\; Z_{c,\,j + c}
  \qquad (c, a, j \in \Z_d),
\]
and the explicit tensor isomorphism
$\Psi = \lean{antiDiagTensorEquiv} : (\Z_d \to F) \otimes_F (\Z_d \to A) \xrightarrow{\;\sim\;}
\Mat_{d\times d}(A)$ sending $u \otimes v$ to $(c, j) \mapsto u_c\,v_{j+c}$.

\begin{theorem}[Pairwise shift-collision characterization and anti-diagonal tensor separation]\label{thm:anti-diagonal}
\formal{OperatorDualityProofs.lean}{pairwiseShiftCollisionCharacterization,antiDiagonalTensorSeparation}
\formal{DualityAndSwapConsequences.lean}{antiDiagTensorEquiv_intertwines_rot,shiftSetCovariance_tensor_finrank}
\formal{OperatorDuality.lean}{PairwiseShiftCollisionCharacterizationSpec,AntiDiagonalTensorSeparationSpec}
Let $F$ be a field, $d \ge 1$, $A$ an $F$-vector space, and $p : \Z_d \to \kappa$.
\begin{enumerate}[label=(\alph*)]
\item \emph{Pairwise shift-collision characterization}: a linear subspace $C \le \Mat_{d\times d}(A)$
  admits deterministic (possibly nonlinear) $p$-shared decoders (\leanname{SharedDecoders.IsSharedCode})
  if and only if for every shift $s \in \Z_d$ and every $Y \in C$ with $\rot_s(Y) \in C$, $Y$
  satisfies pairwise shift covariance (\lean{InPairwiseShiftCovariance}): whenever
  $p(c) = p(c+s)$, $Y_{c+s,\,j} = Y_{c,\,j+s}$ for all $j \in \Z_d$.
\item \emph{Right-factor untwisting of $\rot_t$}: under $\Psi : (\Z_d \to F) \otimes_F (\Z_d \to A)
  \xrightarrow{\sim} \Mat_{d\times d}(A)$, column rotation $\rot_t$ intertwines with the pure
  right-factor coordinate shift $\mathrm{id} \otimes \sigma^t$, where $(\sigma^t v)_a = v_{a+t}$.
\item \emph{Left-factor untwisting of shift-link covariance}: for any shift set $E \subseteq \Z_d$,
  let $\sim_E$ denote the equivalence relation generated by \leanname{SwapRigidity.Link p E} and let
  $U_E(p) \le (\Z_d \to F)$ be the subspace of vectors constant on $\sim_E$-components
  (\lean{leftComponentSubmodule}). Then $\Psi$ maps $U_E(p) \otimes_F (\Z_d \to A)$ onto the
  shift-set covariance subspace $H_E(p)$ (\lean{shiftSetCovarianceSubmodule}), and when
  $\dim_F A < \infty$,
  \[
    \dim_F H_E(p) \;=\; \dim_F U_E(p) \cdot d \cdot \dim_F A.
  \]
\end{enumerate}
\end{theorem}
\begin{outline}
In~(a), determinism of $f_{p(c)}$ on $C$ is equivalent by linearity of $C$ to saying that whenever
$X_1, X_2 \in C$ satisfy $p(c_1) = p(c_2)$ and $\rot_{c_1}(X_1) = \rot_{c_2}(X_2)$, rows $c_1$ of
$X_1$ and $c_2$ of $X_2$ agree; setting $s = c_2 - c_1$ and $Y = X_1 - \rot_{-s}(X_2) \in C$ gives
$\rot_s(Y) = \rot_s(X_1) - X_2 \in C$ and the stated covariance. In~(b),
$\Psi(\mathrm{id}\otimes \sigma^t(u \otimes v))_{c,j} = u_c v_{j+c+t} = \rot_t(\Psi(u\otimes v))_{c,j}$.
In~(c), writing $X_{c,j} = Z_{c, j+c}$ turns $X_{c', j+\delta'} = X_{c, j+\delta}$ under
$c + \delta = c' + \delta'$ into $Z_{c', a} = Z_{c, a}$ for every anti-diagonal $a = c + \delta + j$,
independently across $a \in \Z_d$.
\end{outline}

\begin{proposition}[Mask-certificate floors at $9/16$ and $5/8$ and generic tensor realization]\label{prop:mask-floors}
\formal{DiagnosisAndBarriersProofs.lean}{diagnosis_LinearSharedDecoderBoundsAndMaskFloors}
\formal{DiagnosisAndBridgesConsequences.lean}{diagnosis4_ExplicitSquareCoverAndMaskCertificateBridge}
\formal{DualityAndSwapConsequences.lean}{maskFloor_GenericSubspaceRealization}
\formal{DiagnosisAndBarriers.lean}{Diagnosis_LinearSharedDecoderBoundsAndMaskFloorsSpec}
Let $F$ be a field, $d \ge 1$, and $A$ a finite-dimensional $F$-vector space of dimension
$b := \dim_F A \ge 1$.
\begin{enumerate}[label=(\alph*)]
\item \emph{Universal capacity vs.\ random-mask certificate floors}
  (\cite[Theorems~3.4, 3.5, C.6, and~C.7]{SharedDecoders2026}): every $2$-class linear shared code
  satisfies $\dim_F C \le \frac{41}{64} b d^2 + \frac{5}{3} b d^{3/2}$, and every $M$-class linear
  shared code ($1 \le M, 2M \le d$) has rate $\le 1 - \frac{1}{4\lceil\sqrt{2M}\rceil}$
  (\lean{ceilSqrtTwoMul}); meanwhile, on uniformly random binary masks $p : \Z_d \to \mathrm{Bool}$
  ($\eta \ge 0$), the full-ambient isolation certificate exceeds
  $\bigl(\frac{9}{16} - \frac{\eta}{4} + \frac{1}{2d}\bigr)bd^2$ outside probability
  $(d-1)e^{-2\eta^2 d / 9}$ and the nested-component certificate exceeds
  $\bigl(\frac{5}{8} - \frac{\eta}{4}\bigr)bd^2$ outside probability $(d-1)e^{-\eta^2 d / 2}$.
\item \emph{Single-shift generic tensor realization of the $5/8$ floor}: for any shift $s \in \Z_d$,
  any equal-dimension pair $C_0, C_s \le \Mat_{d\times d}(A)$ with $C_0 \cap C_s \le H_{\{0,s\}}(p)$
  obeys $2\dim_F C_0 \le d^2\dim_F A + \dim_F H_{\{0,s\}}(p)$, and generic anti-diagonal tensor
  subspaces $C_0 = \Psi(V_0 \otimes \top)$, $C_s = \Psi(V_s \otimes \top)$ attain
  $C_0 \cap C_s = H_{\{0,s\}}(p)$ with
  $2\dim_F C_0 + d\dim_F A \ge d^2\dim_F A + \dim_F H_{\{0,s\}}(p)$---reaching rate
  $\ge \frac{5}{8} - \frac{\eta}{4} - \frac{1}{2d}$ whenever
  $\leanname{SharedDecoders.componentCount}(p, \{0,s\}) \ge (\frac{1}{4} - \frac{\eta}{2})d$.
\item \emph{Alternating-mask separation across shifts}: on $d = 2m$ ($m \ge 2$) with alternating
  $p(c) = (c \bmod 2 == 1)$, shift $s = 1$ has $p(c) \ne p(c+1)$ everywhere so
  $U_{\{0,1\}}(p) = \top$ and $H_{\{0,1\}}(p) = \top$ (allowing single-shift rate $1$), whereas
  shift $s = 2$ has $\dim_F U_{\{0,2\}}(p) = 2$ and forces every true shared code $C$ to satisfy
  $2\dim_F C \le (d^2 + 2d)\dim_F A$ (rate $\le \frac{1}{2} + \frac{1}{d}$).
\end{enumerate}
\end{proposition}
\begin{outline}
Part~(a) bridges \cite{SharedDecoders2026}'s universal capacity theorems and Hoeffding random-mask
concentration bounds with explicit integer square-cover arithmetic on \lean{ceilSqrtTwoMul}.
Part~(b) splits $(\Z_d \to F) = U_{\{0,s\}}(p) \oplus W_{\mathrm{comp}}$ and assigns complementary
$\lfloor \dim W_{\mathrm{comp}} / 2 \rfloor$-dimensional halves to $V_0$ and $V_s$ before tensoring
with $(\Z_d \to A)$. Part~(c) checks that odd shifts never link equal parities while shift~$2$ links
all even indices into one component and all odd indices into a second component.
\end{outline}

\subsection{Operator Schmidt Rank separates transpose from Stone's perfect shuffle}\label{sec:osr}

We now turn to \cite[Problem~9.5]{SeventeenthPaper2026}: why can a multi-tape Turing machine execute
Stone's perfect shuffle \cite{Stone1971} in $O(L)$ steps, whereas matrix transposition is conjectured
to require $\Omega(L\log L)$ steps, even though both permute bit addresses of $\Mat_{d\times d}(F^b)$?

View $\lean{GridState}(F, d, b) = (\mathrm{Fin}\,d \to \mathrm{Fin}\,d \to \mathrm{Fin}\,b \to F)$
as the tensor product $F_{\mathrm{row}}^d \otimes_F (F_{\mathrm{col}}^d \otimes_F F^b)$ separating
the \emph{coarse row register} $r \in \mathrm{Fin}\,d$ from the \emph{within-row column-and-lane
register} $(c, \ell) \in \mathrm{Fin}\,d \times \mathrm{Fin}\,b$. An $F$-linear operator $K$ on
$\lean{GridState}(F, d, b)$ has \emph{Operator Schmidt Rank $\le R$}, written $\OSR(K) \le R$
(\lean{HasOperatorSchmidtRankLe}), if there exist $R$ row matrices $A_i \in \Mat_{d\times d}(F)$ and
$R$ within-row maps $B_i \in \End_F(F^d \otimes F^b)$ such that
\[
  (K X)_{r,c,\ell} \;=\; \sum_{i=0}^{R-1} \sum_{r'=0}^{d-1} (A_i)_{r,r'}\,\bigl(B_i(X_{r'})\bigr)_{c,\ell},
\]
and \emph{exact} Operator Schmidt Rank $\OSR(K) = R$ (\lean{HasExactOperatorSchmidtRank}) if $R$ is
minimal.

\begin{lemma}[Submultiplicativity and rank-$1$ locality]\label{lem:osr-submult}
\formal{OperatorDualityProofs.lean}{operatorSchmidtRankSubmultiplicative}
\formal{DualityAndSwapConsequences.lean}{isRowOrColUniform_rankLe_one}
\formal{OperatorDuality.lean}{OperatorSchmidtRankSubmultiplicativeSpec}
Over any field $F$:
\begin{enumerate}[label=(\alph*)]
\item If $\OSR(K_1) \le R_1$ and $\OSR(K_2) \le R_2$, then $\OSR(K_2 \circ K_1) \le R_1 R_2$.
\item Every row-uniform column operator $I_d \otimes B$ (\lean{IsRowUniformColOp}) and every
  column-uniform row operator $A \otimes I_{db}$ (\lean{IsColUniformRowOp}) satisfies
  $\OSR(K) \le 1$.
\end{enumerate}
\end{lemma}
\begin{outline}
Expanding $(K_2 \circ K_1)$ across pairs $(i_2, i_1) \in \mathrm{Fin}\,R_2 \times \mathrm{Fin}\,R_1$
multiplies the row matrices $A_{2,i_2} A_{1,i_1}$ and composes the column operators
$B_{2,i_2} \circ B_{1,i_1}$, using at most $R_1 R_2$ elementary Kronecker terms; row-uniform and
column-uniform maps are single elementary tensors $I_d \otimes B$ and $A \otimes \mathrm{id}$.
\end{outline}

\begin{theorem}[Exact Operator Schmidt Rank: $\OSR(\tau) = d^2$ vs.\ $\OSR(S) = 4$]\label{thm:osr-swap}
\label{thm:osr-shuffle}
\formal{OperatorDualityProofs.lean}{operatorSchmidtRankSwap,operatorSchmidtRankStoneShuffle}
\formal{OperatorDuality.lean}{OperatorSchmidtRankSwapSpec,OperatorSchmidtRankStoneShuffleSpec}
Let $F$ be any field and $b \ge 1$.
\begin{enumerate}[label=(\alph*)]
\item \emph{Matrix transpose} (\lean{gridSwapLinear}): for every $d \ge 1$,
  $\OSR(\tau) = d^2$.
\item \emph{Stone's perfect shuffle and unshuffle} (\lean{stoneShuffleGridLinear},
  \lean{stoneUnshuffleGridLinear}): on $d = 2^m$, $S$ and $S^{-1}$ are mutual inverses with
  $\OSR(S) \le 4$ and $\OSR(S^{-1}) \le 4$ for all $m \ge 0$, and for every $m \ge 1$ ($d = 2^m \ge 2$),
  \[
    \OSR(S) \;=\; \OSR(S^{-1}) \;=\; 4.
  \]
\end{enumerate}
\end{theorem}
\begin{outline}
For~(a), the $d^2$ matrix units $E_{c,r} \otimes E_{r,c}^{\mathrm{lane\,0}}$ decompose $\tau$ into
$d^2$ terms, and evaluating any $R$-term decomposition on row-localized basis inputs exhibits a
spanning set of size $R$ for the $d^2$-dimensional space of $d\times d$ matrices (via linear
independence of the $d^2$ partial-transpose slices), forcing $R \ge d^2$.

For~(b), when $d = 2^m$ ($m \ge 1$), Stone's $1$-bit left cyclic address rotation sends cell
$(r, c)$ to $(r', c')$ where the high bit of $r$ moves into the low bit of $c'$ and the high bit
of $c$ moves into the low bit of $r'$. Splitting on the $2 \times 2 = 4$ parities/halves
$(\text{low bit of } c',\; \text{low bit of } r') \in \{0,1\}^2$ writes $S = \sum_{u,v \in \{0,1\}}
A_{u,v} \otimes B_{u,v}$ in $4$ Kronecker terms, and linear independence of those $4$ non-zero
quadrant-halving/doubling blocks when $2^m \ge 2$ proves no $R' < 4$ suffices (and symmetrically
for $S^{-1}$).
\end{outline}

\begin{theorem}[Pass lower bounds for Stone-shuffle and bounded-OSR pipelines]\label{thm:osr-pass}
\label{cor:general-osr-pass}
\formal{OperatorDualityProofs.lean}{stoneShufflePipelinePassLowerBound}
\formal{DualityAndSwapConsequences.lean}{boundedOSRPipelinePassLowerBound}
\formal{OperatorDuality.lean}{StoneShufflePipelinePassLowerBoundSpec}
Let $F$ be any field and $b \ge 1$.
\begin{enumerate}[label=(\alph*)]
\item On $d = 2^m$ ($m \ge 0$), any linear operator pipeline (\lean{composeGridOps}) composed of
  arbitrarily many row-uniform column operators (\lean{IsRowUniformColOp}), arbitrarily many
  column-uniform row operators (\lean{IsColUniformRowOp}), and $t$ Stone shuffle or unshuffle passes
  that equals $\tau$ satisfies
  \[
    (2^m)^2 \;\le\; 4^t \qquad\text{and hence}\qquad t \;\ge\; m = \log_2 d.
  \]
\item More generally, for any $d \ge 1$ and $R \ge 2$, any pipeline realizing $\tau$ with $t$
  operators of $\OSR \le R$ and arbitrarily many operators of $\OSR \le 1$ satisfies $d^2 \le R^t$
  and $t \ge \dfrac{2\log d}{\log R}$.
\end{enumerate}
\end{theorem}
\begin{outline}
By \Cref{lem:osr-submult} and \Cref{thm:osr-shuffle}(b), the composed pipeline has
$\OSR \le 4^t$ (respectively $\OSR \le R^t$), whereas $\OSR(\tau) = d^2$ by \Cref{thm:osr-swap}(a).
Monotonicity of $\log$ yields $t \ge \log_2 d$ and $t \ge 2\log d / \log R$.
\end{outline}

To complement \Cref{thm:osr-pass} on actual machines, we construct native $O(L)$-time
\leanname{Transpose.TM} implementations of Stone's shuffle, Stone's unshuffle, and within-row shear,
confirming that the $O(1)$-vs.-$\log_2 d$ separation is operationally real on multi-tape Turing
machines:

\begin{proposition}[Native $O(L)$-time Turing machines for shuffle, unshuffle, and row shear]\label{prop:stone-tm}
\formal{DiagnosisAndBarriersProofs.lean}{diagnosis_StoneShuffleAndRowShearMachineComplexity,stoneUnshuffleTM_twoTape_linear}
\formal{DiagnosisAndBarriers.lean}{Diagnosis_StoneShuffleAndRowShearMachineComplexitySpec}
\begin{enumerate}[label=(\alph*)]
\item Stone's shuffle on $d = 2^m$ ($m \ge 1$) is computed in $O(L)$ steps on a $3$-work-tape
  \leanname{Transpose.TM}, Stone's unshuffle on $d = 2^m$ ($m \ge 1$) is computed in $O(L)$ steps on
  a $2$-work-tape \leanname{Transpose.TM} (\lean{stoneUnshuffleTM_twoTape_linear}), and within-row
  cyclic shear $X_{r,c} \mapsto X_{r, (c+r)\bmod d}$ ($d \ge 1$) is computed in $O(L)$ steps on a
  $2$-work-tape \leanname{Transpose.TM}.
\item Over every field $F$, for all $m \ge 1$ ($d = 2^m$), all $b \ge 1$, and every dyadic
  column-bit scale $j \in \mathrm{Fin}\,m$, the commutator $[S, I_{\mathrm{row}} \otimes G_j]$ of a
  single Stone shuffle pass $S$ (\lean{stoneShuffleGridLinear}) with the $j$-th column-bit flip
  involution $G_j$ (\lean{colBitFlipGridLinear}) has exact macroscopic rank
  $\rank_F(S G_j - G_j S) = b(2^m)^2 / 2$ simultaneously across all $m = \log_2 d$ scales
  (cf.\ \cite[Section~4.3, Corollary~6.13, and Remark~6.14]{SeventeenthPaper2026}).
\end{enumerate}
\end{proposition}
\begin{outline}
Because $d$ and $b = \clog{d}$ are not hardcoded into the fixed machine, the shuffle machine first
discovers unary counters for $b$ and $H = d^2/2$ across tapes~1 and~2 in $O(L)$ steps, buffers the
first $H$ input words on tape~0, and interleaves $b$-bit words from tape~0 and the second input half.
The $2$-work-tape unshuffle machine keeps the unary $b$-counter on tape~1, emits even-indexed $b$-bit
words directly from the input tape to the output tape while buffering odd-indexed words on tape~0,
and then replays tape~0 to output. Row shear discovers $d$ and $b$ and cyclically rotates each
$db$-bit row by $rb$ bits. For~(b), $S G_j - G_j S$ pairs off complementary bit-$j$/high-bit cell
orbits of size~$2$ on half the grid entries, giving exact rank $b(2^m)^2 / 2$.
\end{outline}

\subsection{Adversarial controls: row-diagonal gauges and skew one-time pads}\label{sec:osr-controls}

Two sharp controls prevent over-interpreting $\OSR$ outside its domain of validity:
\begin{enumerate}[label=(\roman*)]
\item A $2$-tape machine can execute within-row shear $(K_{\mathrm{shear}} X)_{r,c} = X_{r, c+r}$
  in $O(L)$ steps (\Cref{prop:stone-tm}(a)) because rows never talk to each other, yet $K_{\mathrm{shear}}$
  applies a \emph{different} operator on each row $r$, giving $\OSR(K_{\mathrm{shear}}) = d$! We
  show that replacing $\OSR \le 1$ by the full row-diagonal subalgebra
  $\mathcal{D}_{\mathrm{row}} \otimes \End_F(F^d \otimes F^b)$ (\lean{IsRowDiagonalOp}) preserves
  the $s \ge \log_2 d$ pass lower bound via \emph{output-row source arity}
  (\lean{HasOutputRowSourceArityLe}).
\item For \emph{nonlinear} intermediate encodings on work tapes, cell-wise Shannon mutual information
  with target columns can vanish identically even when $W \oplus X = X^\top$!
\end{enumerate}

\begin{proposition}[Row-shear $\OSR = d$ vs.\ row-diagonal gauge invariance]\label{prop:shear-control}
\formal{OperatorDualityProofs.lean}{control_RowShearAndRowDiagonalAlternation}
\formal{DualityAndSwapConsequences.lean}{rowDiagonal_SourceArityGaugeInvariance}
\formal{OperatorDuality.lean}{Control_RowShearAndRowDiagonalAlternationSpec}
Let $F$ be any field, $d \ge 1$, and $b \ge 1$.
\begin{enumerate}[label=(\alph*)]
\item Within-row shear (\lean{rowShearLinear}) belongs to \lean{IsRowDiagonalOp} and has exact
  Operator Schmidt Rank $\OSR(K_{\mathrm{shear}}) = d$.
\item Output-row source arity (\lean{HasOutputRowSourceArityLe}) is gauge-invariant under two-sided
  composition $D_1 \circ K \circ D_0$ with arbitrary row-diagonal operators $D_0, D_1$.
\item On $d = 2^m$, every alternating product $D_s U_s D_{s-1} \cdots U_1 D_0$
  (\lean{alternatingRowDiagProduct}) of $s + 1$ arbitrary row-diagonal operators $D_i$ and $s$
  Stone shuffle/unshuffle passes $U_i \in \{S, S^{-1}\}$ has output-row source arity $\le 2^s$; if
  it equals $\tau$, then $2^m \le 2^s$ and $s \ge m = \log_2 d$.
\end{enumerate}
\end{proposition}
\begin{outline}
In~(a), $K_{\mathrm{shear}} = \sum_{r=0}^{d-1} E_{r,r} \otimes \sigma^r$, and linear independence of
the $d$ distinct column shifts $\sigma^0, \dots, \sigma^{d-1}$ forces $\OSR = d$. In~(b) and~(c),
row-diagonal operators transform each row in place (multiplying source-row sets by $1$), whereas
each $S$ or $S^{-1}$ reads at most $2$ source rows per output row, bounding source arity by $2^s$;
since row~$0$ of $\tau(X)$ depends on all $2^m$ rows of $X$, $2^m \le 2^s$.
\end{outline}

\begin{proposition}[Skew one-time-pad zero-mutual-information trap]\label{prop:otp-control}
\formal{OperatorDualityProofs.lean}{control_SkewOneTimePadInformationTrap}
\formal{DualityAndSwapConsequences.lean}{uniformShannonMI_eq_zero_of_indep,skewOneTimePad_ShannonMutualInfoZero}
\formal{OperatorDuality.lean}{Control_SkewOneTimePadInformationTrapSpec}
For $d, b \ge 1$, define $W(X) := X^\top \oplus X$ on $\Sigma_b^{d\times d}$ ($\Sigma_b = \{0,1\}^b$,
\lean{skewXorPad}). Then $W(X) \oplus X = X^\top$ for all $X$, every diagonal cell $W_{r,r}$ is
identically $0$, and under uniformly random $X \in \Sigma_b^{d\times d}$, every single cell
$W_{r,c}$ is statistically independent of every target column $X_{:, c'}$---so its discrete Shannon
mutual information (\lean{uniformShannonMI}) satisfies
\[
  I\bigl(W_{r,c}\;;\;X_{:,c'}\bigr) \;=\; 0
  \qquad \text{for all } r, c, c' \in \Z_d.
\]
\end{proposition}
\begin{outline}
When $r = c$, $W_{r,r} = 0$ is deterministic. When $r \ne c$, at least one of the two distinct
matrix entries $X_{c,r}$ and $X_{r,c}$ lies outside column $c'$ (since $(c,r)$ and $(r,c)$ have
distinct column indices $r \ne c$!), acting as an independent uniform one-time pad on the other and
making the joint law of $(W_{r,c}, X_{:,c'})$ a product distribution of zero mutual information.
\end{outline}
