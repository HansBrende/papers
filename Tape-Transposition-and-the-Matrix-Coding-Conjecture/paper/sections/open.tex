\section{What is left}\label{sec:open}

\begin{problem}[The conjecture]\label{prob:conj}
Prove or refute \cref{conj:weak}. A proof decides \textup{(T)} in the negative. A refutation is a family of
two-sided whole-row codes whose deficit at window $U$ falls below the bound of \cref{rem:weak-demand} for all large
sizes; it removes this route and does not give a machine.
\end{problem}

\begin{problem}[Window three]\label{prob:three}
Is $R(3)=\tfrac12$? The known bounds are $\tfrac12\le R(3)\le\tfrac34$. The open relations have spread rows and
offsets. A standing test family is that of the helper maps whose incidence graph is a cubic expander with girth
tending to infinity: at window three no code with a positive asymptotic gain over the pairing rate is known on it,
and no bound below the acyclic one. (The lifts of \cref{thm:E} also have growing girth, at window $3600$, and there
entropy does give a bound below the acyclic one.)
\end{problem}

\begin{problem}[A deficit that does not depend on the window]\label{prob:common}
Find any $\delta>0$ and any class of relations, containing relations with $\sigma(\rho)\ge(1-\eta)d^2$ for every
$\eta>0$, on which every valid polymatroid has value at most $(1-\delta)d^2$.
\end{problem}

\begin{problem}[High-rate codes]\label{prob:highrate}
Can the storage codes of rate tending to one of the fifteenth paper~\cite{HighRate2026}, which amplify those
of~\cite{StorageCodes2026}, be whole-row codes at a window
that depends on the rate only?
\end{problem}

\begin{problem}[Growing windows]\label{prob:growing}
For which functions $w(d)$ does the best rate of a whole-row code of window $w(d)$ on $d\times d$ arrays tend to one?
It does for $w(d)=d^{1/2+\eps}$ (\cref{rem:growing}), and \cref{conj:MCC} says that it does not for bounded~$w$. For
the codes of \cref{prop:matchings} the rate is at most one half in the limit whenever $w(d)=o(d^{1/3})$, even
with mixed matching sizes. For arbitrary codes we know of nothing between bounded
windows and $d^{1/2+\eps}$, for example for $w(d)=\log d$. An argument for the conjecture has to fail for windows of
size $d^{1/2+\eps}$.
\end{problem}

\begin{problem}[Converse]\label{prob:converse}
Does the Tape Coding Conjecture imply any bound on whole-row codes? Does a family of whole-row codes of rate tending
to one, with any further property, yield a causal code on a tape diagram?
\end{problem}
Two questions of earlier papers are untouched by this paper: whether the plain Tape Coding Conjecture and the two
statements of the decomposition of the fourth paper imply one another~\cite[Remark~4.12]{FourthPaper2026}, and the
half-rate questions of the synthesis for a fixed alphabet~\cite[Problem~9.4]{SeventeenthPaper2026}.
