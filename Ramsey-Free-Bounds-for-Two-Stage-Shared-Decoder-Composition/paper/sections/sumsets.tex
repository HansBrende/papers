\section{Shifted sumsets and logarithmic two-stage rank deficits}\label{sec:sumsets}

\subsection{Minimum-shifted monochromatic sumsets}\label{sec:sumsets-step1}

For approximate two-stage compositions at fixed class counts $M_1, M_2 \ge 1$, write
\[
  r \;:=\; \rank_F(\tau - K_2 \circ K_1).
\]
In \cite[Theorem~6.6]{SharedDecoders2026}, extracting a monochromatic progression of length
$\ell = (2M_2 + 1)k - 1$ inside $p_1$ gave $r \ge bk(d - N_0 + 1)/2$ for all $d \ge N_0$ with
$N_0 = \mathsf{W}(M_1, \ell)$ over every field $F$ with $\operatorname{char}(F) \nmid d$
($(d : F) \ne 0$), proving $r/(bd) \to \infty$ as $d \to \infty$. At $M_1 = 1$,
$\mathsf{W}(1, \ell) = \ell$ with $k \approx d/(2(2M_2 + 1))$ already yields a quadratic rank deficit
$\Omega(bd^2 / M_2)$ for all $d \ge 4M_2 + 1$ (\cite[Theorem~6.6]{SharedDecoders2026},
\leanname{SharedDecoders.twoHop_rank_interval_any_field}). For $M_1 \ge 2$, however,
\cite[Theorem~6.6]{SharedDecoders2026} is stated with qualitative $\mathsf{W}(M_1, \ell)$ via the
Hales--Jewett theorem \cite{HalesJewett1963}, and substituting Gowers' van der Waerden bound
\cite{Gowers2001} gives only an iterated-logarithmic growth rate for $r/(bd)$ (moreover, non-wrapping
monochromatic progressions in $\{0, \dots, d-1\}$ under random $2$-colorings have length $O(\log d)$
by a standard first-moment calculation \cite{GRS1990,AlonSpencer2016}, and translate-covering $\Z_d$
by a Behrend set \cite{Behrend1946} limits progressions modulo~$d$ to length $< 3$ at
$M_1 = 2^{O(\sqrt{\log d})}$). Throughout this section, $\log$ denotes the natural logarithm
(\leanname{Real.log}).

Over any field $F$ in which $d$ is invertible ($(d : F) \ne 0$), we pass to a splitting field $E/F$
carrying an injective additive character $\chi : \Z_d \to E^\times$ and replace progressions by
$(n + 1)$-fold monochromatic sumsets at $n = 2M_2 - 1$ (still at frequency width $k = O(\log d)$, but
replacing five-fold exponential van der Waerden thresholds for $M_1 \ge 2$ by K\H{o}v\'ari--S\'os--Tur\'an
density bounds): an \emph{unshifted} $(n+1)$-fold monochromatic
sumset in class $m_1$ consists of a base set $J_0 \subseteq \Z_d$ ($\abs{J_0} = k$) and $n$
natural-number shift sets $B_0, \dots, B_{n-1} \subseteq \N$ ($\abs{B_t} = k + 1$) whose cast sumset in
$\Z_d$ satisfies
\[
  J_0 + B_0 + \cdots + B_{n - 1} \;\subseteq\; p_1^{-1}(m_1).
\]
For any finite $S \subseteq \N$, write $\diam_{\N}(S) := \max S - \min S$ ($0$ if $S = \emptyset$;
\lean{finsetNatDiam}). Shifting each $B_t$ down by $\min B_t$ places $0 \in \widetilde{B}_t$ while
preserving $\diam_{\N}(\widetilde{B}_t) = \diam_{\N}(B_t)$, producing a \emph{shifted} $(n+1)$-fold
monochromatic sumset $(\widetilde{J}_0, \widetilde{B}, \mathrm{diamB})$ ($\abs{\widetilde{J}_0} = k$,
$0 \in \widetilde{B}_t \subseteq [0, \mathrm{diamB}_t] \subseteq \N$, $\abs{\widetilde{B}_t} = k + 1$,
and $\widetilde{J}_0 + \sum_{t=0}^{n-1} \widetilde{B}_t \subseteq p_1^{-1}(m_1)$) so that all partial
subset sums remain in $p_1^{-1}(m_1)$ simultaneously.

\begin{lemma}[Minimum-shifting monochromatic sumsets]\label{lem:sumset-shift}
\formal{SumsetRankDeficitProofs.lean}{unshiftedToShiftedSumset}
\formal{SumsetRankConsequences.lean}{finsetNatDiam_shift_min,unshiftedToShiftedSumset_explicit,unshiftedToShiftedSumset_intrinsicDiam}
Let $d, M_1 \ge 1$ and $p_1 : \Z_d \to \mathrm{Fin}\,M_1$. Given any unshifted monochromatic sumset
$J_0 + B_0 + \cdots + B_{n-1} \subseteq p_1^{-1}(m_1)$ ($J_0 \subseteq \Z_d$, $B_t \subseteq \N$) with
$\abs{J_0} = k$, $\abs{B_t} = k + 1$, and $B_t \subseteq [0, L_0]$, setting
$\widetilde{B}_t := B_t - \min B_t$ and $\widetilde{J}_0 := J_0 + \sum_{t=0}^{n-1} \min B_t$ satisfies
$0 \in \widetilde{B}_t$, $\abs{\widetilde{B}_t} = k + 1$,
$\diam_{\N}(\widetilde{B}_t) = \diam_{\N}(B_t) \le L_0$, $\abs{\widetilde{J}_0} = k$, and
\[
  \widetilde{J}_0 + \sum_{t \in S} \widetilde{B}_t \;\subseteq\; p_1^{-1}(m_1)
  \qquad \text{for every subset } S \subseteq \{0,\dots,n-1\}
\]
(proved directly for the explicit definitions $\widetilde{B}_t, \widetilde{J}_0$ in
\lean{unshiftedToShiftedSumset_explicit}, and packaged with $\mathrm{diamB}_t = \diam_{\N}(B_t)$ at
$n = 2M_2 - 1$ for $M_2 \ge 1$ in \lean{unshiftedToShiftedSumset_intrinsicDiam}).
\end{lemma}
\begin{outline}
Choosing $b_t \in \widetilde{B}_t$ for $t \in S$ and $0 \in \widetilde{B}_t$ for $t \notin S$
rewrites $c_0 + \sum_{t=0}^{n-1} \min B_t + \sum_{t \in S} b_t = c_0 + \sum_{t=0}^{n-1} (b_t + \min B_t)$
with $b_t + \min B_t \in B_t$. Translation by $-\min B_t$ on $\N_{\ge \min B_t}$ preserves both
cardinality and $\max - \min$.
\end{outline}

\subsection{Arbitrary-subset Vandermonde pullback and sparse annihilating polynomials}\label{sec:sumsets-poly}

On an arbitrary $k$-element subset $J_0 \subseteq \Z_d$, the character-evaluation matrix
$\bigl(\chi(\nu \cdot c)\bigr)_{c \in J_0,\; 0 \le \nu < k}$ is a $k \times k$ Vandermonde matrix in
the $k$ distinct scalars $z_c = \chi(c)$ ($c \in J_0$), since $\chi(\nu \cdot c) = \chi(c)^\nu$.
Thus $J_0$ need not be an interval or progression for stacked evaluation on the frequency-$< k$ band
to be injective. Following \cite[Section~6]{SharedDecoders2026}, for a partition
$p_2 : \Z_d \to \mathrm{Fin}\,M_2$ and $k \le d$, let $H := {}$\leanname{SharedDecoders.bandRowSpace}
$(\chi, p_2, k) \le (E^b)^{\Z_d}$ be the $E$-span of the $b M_2 k$ class-modulated character vectors
$i \mapsto \mathbf{1}_{p_2(i) = m_2}\,\chi(\nu i)\,e_\ell$ ($m_2 \in \mathrm{Fin}\,M_2$,
$0 \le \nu < k$, $\ell \in \mathrm{Fin}\,b$, so $\dim_E H \le b M_2 k$), and let
\leanname{SharedDecoders.ColumnAmbient}$(H) = (\mathrm{Unit} \to \Z_d \to H) \cong (\Z_d \to H)$
embed into $\Mat_{d\times d}(E^b)$ via $(\iota_H x)_{i, c} = (x_{(), c})_i$
(\leanname{SharedDecoders.columnAmbientEmbed}), with column shift $(T_{\mathrm{band}} x)_{(), c} =
x_{(), c+1}$ (\leanname{SharedDecoders.channelRotation} at shift~$1$) satisfying
$\iota_H(T_{\mathrm{band}} x) = \rot_1(\iota_H x)$.

\begin{lemma}[Arbitrary-subset Vandermonde injectivity and stacked band pullback]\label{lem:vandermonde-subset}
\formal{SumsetRankDeficitProofs.lean}{vandermondeArbitrarySubsetInjective,stackedBandPullbackInjective}
Let $d \ge 1$, let $E$ be a field with an injective additive character $\chi : \Z_d \to E^\times$, and
let $J_0 \subseteq \Z_d$ satisfy $\abs{J_0} = k \le d$.
\begin{enumerate}[label=(\alph*)]
\item For any $E$-vector space $V$, degree-$< k$ character evaluation
  $a \mapsto \bigl(\sum_{\nu=0}^{k-1} \chi(\nu c)\,a_\nu\bigr)_{c \in J_0}$
  (\lean{charBandEval}) is injective on $V^k$.
\item Let $M_1, M_2 \ge 1$, $p_1 : \Z_d \to \mathrm{Fin}\,M_1$, and
  $p_2 : \Z_d \to \mathrm{Fin}\,M_2$. For any two $E$-linear class-uniform stages $K_1, K_2$ on
  $\Mat_{d\times d}(E^b)$ with $J_0 \subseteq p_1^{-1}(m_1)$ and stage-1 class-$m_1$ decoder
  $f_{1,m_1}$, the frequency-$< k$ band space $H$ (\leanname{SharedDecoders.bandRowSpace} at
  $(\chi, p_2, k)$, with $\dim_E H \le b M_2 k$) carries a pullback subspace
  $W \le \leanname{SharedDecoders.ColumnAmbient}(H)$ satisfying
  \[
    \dim_E W + \rank_E(\tau - K_2 \circ K_1) \;\ge\; b k d
  \]
  such that the stacked evaluation $x \mapsto \bigl(f_{1,m_1}(\rot_c(\iota_H x))\bigr)_{c \in J_0}$
  is injective on $W$.
\end{enumerate}
\end{lemma}
\begin{outline}
In~(a), $\chi(\nu c) = \chi(c)^\nu$, and injectivity of $\chi$ makes $\{\chi(c) : c \in J_0\}$
$k$ distinct scalars in $E$, so a degree-$< k$ vector-valued polynomial vanishing on $J_0$ is zero.
For~(b), parameterize intermediate frequency-$< k$ band states by
$E_{\mathrm{band}}(u)_{i, c} := \sum_{\nu=0}^{k-1} \chi(\nu c)\,u_{\nu, i}$ ($b k d$ dimensions,
injective since $k \le d$). Pulling back $\mathrm{id} - K_1 \circ \tau \circ K_2$ along
$E_{\mathrm{band}}$ cuts out $Q := \ker\bigl((\mathrm{id} - K_1 \circ \tau \circ K_2) \circ
E_{\mathrm{band}}\bigr)$ of dimension $\dim_E Q \ge b k d - \rank_E(\tau - K_2 \circ K_1)$, and
$u \mapsto \tau K_2(E_{\mathrm{band}}(u))$ embeds $Q$ injectively into
$\iota_H\bigl(\leanname{SharedDecoders.ColumnAmbient}(H)\bigr)$ as $W$ ($\dim_E W = \dim_E Q$). For
$x \in W$ coming from $u \in Q$, $X := \iota_H(x) = \tau K_2(E_{\mathrm{band}}(u))$ has intermediate
state $K_1 X = E_{\mathrm{band}}(u)$ whose column $c$ is $\sum_{\nu=0}^{k-1} \chi(\nu c)\,u_\nu$; if
$f_{1,m_1}(\rot_c(X)) = (K_1 X)_{:,c} = 0$ for all $c \in J_0 \subseteq p_1^{-1}(m_1)$, part~(a)
gives $u = 0$, hence $K_1 X = 0$ and $X = \tau K_2 K_1 X = 0$ ($x = 0$).
\end{outline}

Each summand $\widetilde{B}_t$ of size $k + 1$ provides $k + 1$ monomials against $k$ frequencies
$\nu \in \{0, \dots, k-1\}$, guaranteeing a nonzero sparse polynomial supported in $\widetilde{B}_t$
that annihilates the frequency-$< k$ band and has degree at most $\diam_{\N}(\widetilde{B}_t)$.

\begin{lemma}[Sparse sumset annihilating polynomial]\label{lem:sumset-poly}
\formal{SumsetRankDeficitProofs.lean}{sumsetAnnihilatingPolynomial,cyclicChannelShift_aeval_ker_finrank_le}
\formal{SumsetRankConsequences.lean}{sumsetAnnihilatingPolynomial_bandRowSpace}
Let $d \ge 1$, let $E$ be a field with an additive character $\chi : \Z_d \to E^\times$, and
let $\widetilde{B} \subseteq \N$ satisfy $0 \in \widetilde{B}$, $\abs{\widetilde{B}} = k + 1$, and
$\max \widetilde{B} \le L_0$. Then there exists a nonzero polynomial $P \in E[Z]$ supported in
$\widetilde{B}$ with $\deg P \le \diam_{\N}(\widetilde{B}) \le L_0$ such that
$P\bigl(\chi(\nu)\bigr) = \sum_{x \in \widetilde{B}} P_x \chi(\nu x) = 0$ for all $0 \le \nu < k$.
Moreover, for any $q \in \N$, on the $q$-channel cyclic shift module $(\Z_d \to E^q)$ with unit shift
$(T_1 w)_c = w_{c+1}$ one has
\[
  \dim_E \ker P(T_1) \;\le\; q\,\deg P \;\le\; q\,L_0,
\]
and on \leanname{SharedDecoders.ColumnAmbient} over \leanname{SharedDecoders.bandRowSpace} for
$(\chi, p_2, k)$ (where $\dim_E H \le b M_2 k$) one has
$\dim_E \ker P(T_{\mathrm{band}}) \le b M_2 k L_0$.
\end{lemma}
\begin{outline}
The homogeneous linear system $\sum_{x \in \widetilde{B}} P_x \chi(\nu x) = 0$ ($0 \le \nu < k$) has
$k$ equations in $k + 1$ unknowns, hence a nonzero solution $P$ supported in $\widetilde{B}$ of
degree $\le \max\widetilde{B} = \diam_{\N}(\widetilde{B}) \le L_0$. On $(\Z_d \to E^q)$, the linear
recurrence $P(T_1)w = 0$ determines $w$ from at most $\deg P$ coordinates once $\deg P \le d$ (and
trivially $\dim \le q d \le q \deg P$ when $\deg P > d$).
\end{outline}

\subsection{Sumset rank bound, K\H{o}v\'ari--S\'os--Tur\'an extraction, and Theorem 2B}\label{sec:thm2b}

\begin{theorem}[Two-stage sumset recurrence rank bound]\label{thm:sumset-rank}
\formal{SumsetRankDeficitProofs.lean}{sumsetTwoHopRankBound}
Let $F$ be a field with $(d : F) \ne 0$, let $M_1, M_2, b \ge 1$, and let $1 \le k \le d$. Suppose
$p_1 : \Z_d \to \mathrm{Fin}\,M_1$ admits a shifted $(2M_2)$-fold monochromatic sumset
$(\widetilde{J}_0, \widetilde{B}, \mathrm{diamB})$ in class $m_1$ with
$\sum_{t=0}^{2M_2 - 2} \mathrm{diamB}_t \le d$. Then for every $p_2 : \Z_d \to \mathrm{Fin}\,M_2$
and every pair of $F$-linear class-uniform maps $K_1, K_2$ on $\Mat_{d\times d}(F^b)$, writing
$r := \rank_F(\tau - K_2 \circ K_1)$,
\[
  2r \;\ge\; b k \Bigl(d - \sum_{t=0}^{2M_2 - 2} \mathrm{diamB}_t\Bigr),
  \qquad\text{and if } \mathrm{diamB}_t \le \lfloor d/(4M_2)\rfloor \text{ for all } t,\quad
  4r \;\ge\; b k d.
\]
\end{theorem}
\begin{outline}
Since $(d : F) \ne 0$, $Z^d - 1$ is separable over $F$; passing to a cyclotomic splitting field
$E/F$ satisfies $\rank_E(\tau - K_2 \circ K_1) \le \rank_F(\tau - K_2 \circ K_1)$
(\leanname{SharedDecoders.stateExtension_rank_le}, so any lower bound over $E$ transfers to $F$) and
furnishes an injective character $\chi : \Z_d \to E^\times$.
Let $W \le A_{\mathrm{band}}$ ($\dim_E A_{\mathrm{band}} \le b M_2 k d$) be the pullback subspace
from \Cref{lem:vandermonde-subset}(b) on $\widetilde{J}_0$, with $\dim_E W \ge b k d - r$, and let $P_t$
be the sparse annihilating polynomial on $\widetilde{B}_t$ from \Cref{lem:sumset-poly}. Because
$P_t$ annihilates the frequency-$< k$ band while every shift in
$\widetilde{J}_0 + \sum_{s \le t} \widetilde{B}_s$ lies in $p_1^{-1}(m_1)$, $P_t(T_{\mathrm{band}})$ maps the
$t$-th partial sumset span $U_t$ into the kernel of stacked $\widetilde{J}_0$-evaluation, which
meets $W$ trivially. Each of the $2M_2 - 1$ steps therefore adds at least
$\dim_E W - b M_2 k\,\mathrm{diamB}_t$ dimensions inside $A_{\mathrm{band}}$:
\[
  2M_2 \dim_E W - b M_2 k \sum_{t=0}^{2M_2 - 2} \mathrm{diamB}_t \;\le\; \dim_E A_{\mathrm{band}}
  \;\le\; b M_2 k d.
\]
Substituting $\dim_E W \ge b k d - r$ and dividing by $M_2$ yields
$2r \ge bk\bigl(d - \sum_t \mathrm{diamB}_t\bigr)$, and $\mathrm{diamB}_t \le \lfloor d/(4M_2)\rfloor$
gives $\sum_t \mathrm{diamB}_t \le d/2$, hence $4r \ge b k d$.
\end{outline}

\begin{lemma}[Iterated K\H{o}v\'ari--S\'os--Tur\'an monochromatic sumset extraction]\label{lem:kst-sumset}
\formal{SumsetRankDeficitProofs.lean}{iteratedKovariSosTuranSumset}
For every $M_1, M_2 \ge 1$, there exist $0 < c_{\mathrm{lo}} \le C_{\mathrm{hi}}$ and $D_0 \in \N$
such that for all $d \ge D_0$, every $M_1$-coloring $p_1 : \Z_d \to \mathrm{Fin}\,M_1$ contains an
unshifted (and hence shifted) $(2M_2)$-fold monochromatic sumset in $\Z_d$ whose $2M_2 - 1$ shift
summands $B_t$ (and $\widetilde{B}_t$) lie in $[0, \lfloor d/(4M_2)\rfloor] \subseteq \N$ at width
scale $1 \le k \le d$ satisfying
\[
  c_{\mathrm{lo}}\,(\log d)^{\frac{1}{2M_2 - 1}} \;\le\; k \;\le\;
  C_{\mathrm{hi}}\,(\log d)^{\frac{1}{2M_2 - 1}}.
\]
\end{lemma}
\begin{outline}
Start from the densest color class $A_0 \subseteq \{0,\dots,d-1\}$ ($\abs{A_0} \ge \lceil d/M_1\rceil$).
At each of the $2M_2 - 1$ steps, partition $\{0,\dots,d-1\}$ into $4M_2$ intervals of width at most
$\lfloor d/(4M_2)\rfloor + 1$, pick the densest bucket $I \cap A$ of size
$m \ge \lceil\abs{A}/(4M_2)\rceil$, split $I \cap A$ into its lower $\lfloor m/2\rfloor$ and upper
$\lceil m/2\rceil$ elements, and apply the K\H{o}v\'ari--S\'os--Tur\'an double count
\cite{KovariSosTuran1954} to $(k+1)$-tuples of differences
$a_{\mathrm{hi}} - a_{\mathrm{lo}} \in [0, \lfloor d/(4M_2)\rfloor]$ (so $a_{\mathrm{lo}} + b =
a_{\mathrm{hi}} \in A \subseteq [0, d)$ never wraps modulo~$d$). Iterating $2M_2 - 1$ times extracts
$k$ base elements and $2M_2 - 1$ shift sets of size $k + 1$ at density cost exponential in
$k^{2M_2 - 1}$, balancing at $k = \Theta_{M_1,M_2}\bigl((\log d)^{1/(2M_2-1)}\bigr)$.
\end{outline}

\begin{theorem}[Theorem 2B: log-power two-stage rank deficit over admissible fields]\label{thm:2b}
\formal{SumsetRankDeficitProofs.lean}{theorem2B_TwoStageLogPowerRankDeficit}
\formal{SumsetRankConsequences.lean}{theorem2B_ExplicitConstant,theorem2BExplicitConst_pos,theorem2BExplicitThreshold_eq_pow}
For all integers $M_1, M_2 \ge 1$, define the explicit constant and dimension threshold
(\lean{theorem2BExplicitConst}, \lean{theorem2BExplicitThreshold}, with natural logarithm
$\log = \leanname{Real.log}$)
\begin{align*}
  c(M_1, M_2) &\;:=\; \frac{1}{144\,\bigl(\log(16 M_1 M_2)\bigr)^{\frac{1}{2M_2 - 1}}} \;>\; 0, \\
  D_0(M_1, M_2) &\;:=\; \max\!\Bigl(4M_2,\; \bigl\lceil \exp\bigl(36^{2M_2-1}\log(16 M_1 M_2)\bigr)\bigr\rceil + 2\Bigr) \\
  &\;=\; \max\!\Bigl(4M_2,\; (16 M_1 M_2)^{36^{2M_2 - 1}} + 2\Bigr).
\end{align*}
Then for every field $F$, every $d \ge D_0(M_1,M_2)$ with $(d : F) \ne 0$, every lane width
$b \ge 1$, and every pair of $F$-linear class-uniform stages $K_1, K_2$ on $\Mat_{d\times d}(F^b)$
with $M_1$ and $M_2$ classes,
\[
  \rank_F(\tau - K_2 \circ K_1)
  \;\ge\;
  c(M_1, M_2)\,b\,d\,(\log d)^{\frac{1}{2M_2 - 1}}.
\]
\end{theorem}
\begin{outline}
Instantiate the K\H{o}v\'ari--S\'os--Tur\'an chain of \Cref{lem:kst-sumset} at
$k = \bigl\lfloor \bigl(\frac{\log d}{18^{2M_2-1}\log(16M_1 M_2)}\bigr)^{1/(2M_2-1)}\bigr\rfloor \ge 2$
for $d \ge D_0(M_1, M_2)$, and substitute into $4r \ge b k d$ from \Cref{thm:sumset-rank}.
\end{outline}

\begin{remark}[Comparison with {\cite[Theorem~6.6]{SharedDecoders2026}}, asymptotic threshold, and characteristic restriction]\label{rem:2b-threshold}
Three qualifications on \Cref{thm:2b} should be noted explicitly:
\begin{enumerate}[label=(\alph*)]
\item \emph{Parameter comparison}: \Cref{thm:2b} improves on \cite[Theorem~6.6]{SharedDecoders2026}
  only when $M_1 \ge 2$ and $d$ is sufficiently large. At $M_1 = 1$, $\mathsf{W}(1, \ell) = \ell$
  already gives a quadratic rank deficit $\Omega(bd^2 / M_2)$ for all $d \ge 4M_2 + 1$
  (\cite[Theorem~6.6]{SharedDecoders2026}, \leanname{SharedDecoders.twoHop_rank_interval_any_field}).
  At $M_1 = 2$ and $M_2 = 1$, \cite[Theorem~6.6]{SharedDecoders2026} at $k = 2$ with
  $\mathsf{W}(2, 5) = 178$ \cite{StevensShantaram1978} gives $r \ge b(d - 177)$, which \Cref{thm:2b}
  ($c(2,1) = \frac{1}{144\log 32}$) beats only for $d \gtrsim 10^{216}$. For fixed $M_1 \ge 2$ as
  $d \to \infty$, however, \Cref{thm:2b} replaces the iterated-logarithmic growth rates obtained from
  Gowers' van der Waerden bound \cite{Gowers2001} by $(\log d)^{1/(2M_2 - 1)}$.
\item \emph{Threshold size}: the formalized threshold $D_0(M_1, M_2)$ is purely asymptotic: already
  at $M_1 = M_2 = 1$ one has $D_0(1,1) = 16^{36} + 2 = 2^{144} + 2 \approx 2.23 \times 10^{43}$, and
  the exponent $36^{2M_2 - 1}$ grows exponentially in $M_2$.
\item \emph{Characteristic}: like \cite[Theorem~6.6]{SharedDecoders2026} (and unlike \Cref{thm:2a}),
  \Cref{thm:2b} requires $(d : F) \ne 0$ (excluding $\F_2$ at $d = 2^m$) so that $\Z_d$ admits $d$
  distinct character values.
\end{enumerate}
\end{remark}

\begin{corollary}[Single-class second stage: $\Omega(b\,d\log d)$ rank deficit]\label{cor:2b-m2-one}
\formal{SumsetRankConsequences.lean}{theorem2B_OneSecondStage_OmegaLog}
When $M_2 = 1$, for every $M_1 \ge 1$ there exist $c > 0$ and $D_0 \ge 2$ such that for every field
$F$, every $d \ge D_0$ with $(d : F) \ne 0$, every $b \ge 1$, and every $M_1$-class $K_1$ and
$1$-class $K_2$ on $\Mat_{d\times d}(F^b)$,
\[
  \rank_F(\tau - K_2 \circ K_1) \;\ge\; c\,b\,d\log d.
\]
\end{corollary}
\begin{outline}
Specialize \Cref{thm:2b} to $M_2 = 1$, where $\frac{1}{2M_2 - 1} = 1$.
\end{outline}

\subsection{Single-band witness without sumset iteration at \texorpdfstring{$d = 12$}{d = 12}}\label{sec:d12-band}

\Cref{thm:sumset-rank} iterates the sparse annihilating-polynomial step across all $2M_2 - 1$ summands
of a $(2M_2)$-fold monochromatic sumset rather than stopping at a single half-density subset
$J \subseteq \Z_d$ and a single $k = 2$ band (with one fixed shift-covariant linear readout $L$
applied to $T_c w$ for $c \in J$). The following $4$-channel witness at $d = 12$ ($q = b M_2 k = 4$)
shows that under the text-only parameter reading $(b, M_2) = (1, 2)$ ($bkd = 24$, whereas in the
formalized $(b, M_2) = (2, 1)$ bridge one has $bkd = 48$ and $24 = \frac{1}{2}bkd$), a single
half-density band without any annihilating-polynomial step admits $bkd = 24$-dimensional
shift-covariant solutions ($r \ge 0$), so the annihilating-polynomial steps cannot all be dropped.

\begin{example}[Shift-covariant $4$-channel band witness at $d = 12$ over $\C$]\label{ex:d12-band}
\formal{SumsetRankDeficitProofs.lean}{control_D12SingleClassAbstractBand}
\partformal{SumsetRankConsequences.lean}{exists_primitive_twelfth_root_complex,d12AddChar_injective_and_primitive,control_D12_UnconditionalExists,control_D12_BandRowSpaceChannelEquiv}{the alternative $(b = 1, M_2 = 2)$ parameter reading of $q = b M_2 k = 4$ (Lean proves the shift-equivariant \leanname{SharedDecoders.bandRowSpace} bridge for $b = 2, M_2 = 1, k = 2$, where $24 = kd$ is half of $bkd = 48$).}
At $d = 12$ and $k = 2$ on the $4$-channel cyclic module $(\Z_{12} \to \C^4)$ ($q = 4$, identified
shift-equivariantly via \lean{control_D12_BandRowSpaceChannelEquiv} with
\leanname{SharedDecoders.ColumnAmbient} over \leanname{SharedDecoders.bandRowSpace} for
$(\chi, p_0, 2)$ on the single-class partition $p_0 : \Z_{12} \to \mathrm{Fin}\,1$ at
$b = 2, M_2 = 1, k = 2$, where $L$ reads $12$ scalar coordinates per shift and $24 = kd = \frac{1}{2}bkd$;
in text, $q = 4$ also matches the dimension bound $\dim H \le b M_2 k$ at $b = 1, M_2 = 2, k = 2$
where $24 = bkd$), let $J_{12} = \{0, 1, 3, 4, 7, 9\} \subseteq \Z_{12}$ ($\abs{J_{12}} = 6$). With
$\zeta_{12} = \exp(2\pi i / 12) \in \C$, there exist a fixed linear readout
$L : (\Z_{12} \to \C^4) \to \C^{12}$ and a $24$-dimensional subspace $W \le (\Z_{12} \to \C^4)$ on
which stacked shift-covariant evaluation $w \mapsto (L(T_c w))_{c \in J_{12}}$ is injective and
realizes every pair $(a_0, a_1) \in (\C^{12})^2$ as $L(T_c w) = a_0 + \zeta_{12}^c a_1$ for all
$c \in J_{12}$.
\end{example}
\begin{outline}
Verify that $\zeta_{12} = \exp(2\pi i / 12)$ has exact multiplicative order~$12$ in $\C$, construct
the explicit $24$-dimensional $4$-channel encoding whose shifts on $J_{12}$ satisfy
$L(T_c w) = a_0 + \zeta_{12}^c a_1$ for one fixed linear map $L$, and identify $(\Z_{12} \to \C^4)$
shift-equivariantly with \leanname{SharedDecoders.ColumnAmbient} over
\leanname{SharedDecoders.bandRowSpace} for $(\chi, p_0, 2)$ at $b = 2, M_2 = 1, k = 2$. Under the
text-only parameter reading $(b, M_2) = (1, 2)$ ($24 = bkd$, so the band alone gives only $r \ge 0$),
a single half-density shift set $J_{12}$ without any annihilating-polynomial step admits
$bkd = 24$-dimensional shift-covariant solutions, showing in that reading that the
annihilating-polynomial steps in \Cref{thm:sumset-rank} cannot all be dropped
(note that because $q = 4$ corresponds to $b M_2 = 2$ rather than $b = M_2 = 1$, this example does not
provide a $b = 1, M_1 = M_2 = 1$ matrix factorization).
\end{outline}
