\section{Sumset polynomial recurrences and log-power rank deficits}\label{sec:sumsets}

\subsection{From progressions to shifted monochromatic sumsets}\label{sec:sumsets-step1}

\Cref{thm:2a} rules out \emph{exact} factorizations $K_2 \circ K_1 = \tau$ up to exponentially
growing $M_1$ (at fixed $M_2$), yet for \emph{approximate} factorizations with constant class counts
$M_1, M_2$---the regime relevant when trace cuts introduce $o(d^2)$ error---bounding the rank error
\[
  r \;:=\; \rank_F(\tau - K_2 \circ K_1)
\]
via Hales--Jewett (\cite[Theorem~6.6]{SharedDecoders2026}) gave only an inverse-Ackermann/Hales--Jewett
sub-iterated-logarithmic bound, because progressions in $2$-colorings have length at most $O(\log d)$
and single-difference common-message arguments lose the entire remainder once $r \ge bd/\ell$.

Over any field $F$ in which $d$ is invertible ($(d : F) \ne 0$), we pass to a splitting field $E/F$
carrying an injective additive character $\chi : \Z_d \to E^\times$ and replace progressions by
\emph{$(2M_2)$-fold monochromatic sumsets}
\[
  J_0 + B_1 + \cdots + B_{2M_2 - 1} \;\subseteq\; p_1^{-1}(m_1),
  \qquad
  \abs{J_0} = k,\quad \abs{B_t} = k + 1.
\]
For any finite $S \subseteq \N$, write $\diam_{\N}(S) := \max S - \min S$ ($0$ if $S = \emptyset$;
\lean{finsetNatDiam}). Shifting each $B_t$ down by $b_{t,0} = \min B_t$ places $0 \in \widetilde{B}_t$
while preserving $\diam_{\N}(\widetilde{B}_t) = \diam_{\N}(B_t)$, so \emph{all} partial subset sums
land inside $p_1^{-1}(m_1)$ simultaneously.

\begin{lemma}[Minimum-shifting monochromatic sumsets and intrinsic diameter]\label{lem:sumset-shift}
\formal{SumsetRankDeficitProofs.lean}{unshiftedToShiftedSumset}
\formal{SumsetRankConsequences.lean}{finsetNatDiam_shift_min,unshiftedToShiftedSumset_intrinsicDiam}
\formal{SumsetRankDeficit.lean}{UnshiftedToShiftedSumsetSpec}
Given any unshifted monochromatic sumset $J_0 + B_1 + \cdots + B_{n} \subseteq p_1^{-1}(m_1)$ with
$\abs{J_0} = k$, $\abs{B_t} = k + 1$, and $B_t \subseteq [0, L_0]$, setting
$\widetilde{B}_t := B_t - \min B_t$ and $\widetilde{J}_0 := J_0 + \sum_{t=1}^n \min B_t$ satisfies
$0 \in \widetilde{B}_t$, $\abs{\widetilde{B}_t} = k + 1$,
$\diam_{\N}(\widetilde{B}_t) = \diam_{\N}(B_t) \le L_0$, $\abs{\widetilde{J}_0} = k$, and
\[
  \widetilde{J}_0 + \sum_{t \in S} \widetilde{B}_t \;\subseteq\; p_1^{-1}(m_1)
  \qquad \text{for every subset } S \subseteq \{1,\dots,n\}.
\]
\end{lemma}
\begin{outline}
Choosing $b_t \in \widetilde{B}_t$ for $t \in S$ and $0 \in \widetilde{B}_t$ for $t \notin S$
rewrites $c_0 + \sum_t \min B_t + \sum_{t \in S} b_t = c_0 + \sum_{t=1}^n (b_t + \min B_t)$ with
$b_t + \min B_t \in B_t$. Translation by $-\min B_t$ on $\N_{\ge \min B_t}$ preserves cardinality
and $\max - \min$.
\end{outline}

\subsection{Arbitrary-subset Vandermonde pullback and sparse annihilating polynomials}\label{sec:sumsets-poly}

Why did earlier Fourier/band arguments require $J_0$ to be a contiguous interval or arithmetic
progression? Because standard Vandermonde matrices evaluate $1, z, \dots, z^{k-1}$ at geometric
progressions $\zeta^{c_0 + j s}$! Over an arbitrary $k$-element base set $\widetilde{J}_0 \subseteq
\Z_d$, the character-evaluation matrix $\bigl(\chi(\nu \cdot c)\bigr)_{c \in \widetilde{J}_0,\;
0 \le \nu < k}$ is \emph{already} a $k \times k$ Vandermonde matrix in the $k$ distinct nodes
$z_c = \chi(c)$ ($c \in \widetilde{J}_0$), because $\chi(\nu \cdot c) = \chi(c)^\nu$!

\begin{lemma}[Arbitrary-subset Vandermonde injectivity and stacked band pullback]\label{lem:vandermonde-subset}
\label{lem:band-pullback}
\formal{SumsetRankDeficitProofs.lean}{vandermondeArbitrarySubsetInjective,stackedBandPullbackInjective}
\formal{SumsetRankDeficit.lean}{VandermondeArbitrarySubsetInjectiveSpec,StackedBandPullbackInjectiveSpec}
Let $E$ be a field with an injective additive character $\chi : \Z_d \to E^\times$, and let
$J_0 \subseteq \Z_d$ satisfy $\abs{J_0} = k \le d$.
\begin{enumerate}[label=(\alph*)]
\item For any $E$-vector space $V$, degree-$< k$ character evaluation
  $a \mapsto \bigl(\sum_{\nu=0}^{k-1} \chi(\nu c)\,a_\nu\bigr)_{c \in J_0}$
  (\lean{charBandEval}) is injective on $V^k$.
\item For any two $E$-linear class-uniform stages $K_1, K_2$ on $\Mat_{d\times d}(E^b)$ with
  $J_0 \subseteq p_1^{-1}(m_1)$ and stage-1 class-$m_1$ decoder $f_{1,m_1}$, the frequency-$< k$
  band space $H$ (\leanname{SharedDecoders.bandRowSpace} at $(\chi, p_2, k)$, with $\dim_E H \le b M_2 k$)
  carries a pullback subspace $W \le \leanname{SharedDecoders.ColumnAmbient}(H)$ satisfying
  \[
    \dim_E W + \rank_E(\tau - K_2 \circ K_1) \;\ge\; b k d
  \]
  such that the stacked evaluation $x \mapsto \bigl(f_{1,m_1}(\rot_c(\iota_H x))\bigr)_{c \in J_0}$
  is injective on $W$.
\end{enumerate}
\end{lemma}
\begin{outline}
In~(a), $\chi(\nu c) = \chi(c)^\nu$, and injectivity of $\chi$ makes $\{\chi(c) : c \in J_0\}$
$k$ distinct scalars in $E$; a degree-$< k$ vector-valued polynomial vanishing at $k$ distinct
points is identically zero. Part~(b) restricts $\tau - K_2 \circ K_1$ to inputs whose columns lie
in the $k$-frequency character subspace ($b k d$ dimensions), intersects with
$\ker(\tau - K_2 \circ K_1)$ to lose at most $r = \rank_E(\tau - K_2 \circ K_1)$ dimensions, and
uses~(a) to see that if $K_1 X$ vanishes on columns $c \in J_0$, then $(K_2 K_1 X)_{:,c} =
(\tau X)_{:,c} = 0$ for all $c \in J_0$, forcing all $k$ band coefficients of $X$ to vanish.
\end{outline}

Next, each summand $\widetilde{B}_t$ of size $k + 1$ supplies $k + 1$ monomials against only $k$
frequencies $\nu \in \{0, \dots, k-1\}$, guaranteeing a \emph{nonzero} sparse polynomial supported
on $\widetilde{B}_t$ that annihilates the entire frequency-$< k$ band while keeping small degree
$\le \diam_{\N}(\widetilde{B}_t)$!

\begin{lemma}[Sparse sumset annihilating polynomial]\label{lem:sumset-poly}
\formal{SumsetRankDeficitProofs.lean}{sumsetAnnihilatingPolynomial}
\formal{SumsetRankConsequences.lean}{sumsetAnnihilatingPolynomial_bandRowSpace}
\formal{SumsetRankDeficit.lean}{SumsetAnnihilatingPolynomialSpec}
Let $\chi : \Z_d \to E^\times$ be an injective additive character and let $\widetilde{B} \subseteq \N$
satisfy $0 \in \widetilde{B}$, $\abs{\widetilde{B}} = k + 1$, and $\max \widetilde{B} \le L_0$
(equivalently $\diam_{\N}(\widetilde{B}) \le L_0$). Then there exists a nonzero polynomial
$P \in E[X]$ supported in $\widetilde{B}$ with $\deg P \le \diam_{\N}(\widetilde{B}) \le L_0$ such
that $\sum_{x \in \widetilde{B}} P_x \chi(\nu x) = 0$ for all $0 \le \nu < k$, and on any
$q$-channel cyclic shift module $(\Z_d \to E^q)$ (in particular on
$\leanname{SharedDecoders.ColumnAmbient}(\leanname{SharedDecoders.bandRowSpace}(\chi,p_2,k))$, where
$q \le b M_2 k$),
\[
  \dim_E \ker P(T_1) \;\le\; q\,\deg P \;\le\; b M_2 k L_0.
\]
\end{lemma}
\begin{outline}
The homogeneous linear system $\sum_{x \in \widetilde{B}} P_x \chi(\nu x) = 0$ ($0 \le \nu < k$) has
$k$ equations in $\abs{\widetilde{B}} = k + 1$ unknowns, hence a nonzero solution $P$ supported in
$\widetilde{B}$ of degree $\le \max\widetilde{B} = \diam_{\N}(\widetilde{B}) \le L_0$. On a
$q$-channel cyclic sequence $w : \Z_d \to E^q$, $P(T_1)w = 0$ fixes $w$ uniquely from any $\deg P$
consecutive coordinates once $\deg P \le d$ (and trivially $\dim \le q d \le q \deg P$ if
$\deg P > d$), giving $\dim_E\ker P(T_1) \le q\,\deg P$.
\end{outline}

\subsection{Two-stage sumset rank bound, K\H{o}v\'ari--S\'os--Tur\'an extraction, and Theorem 2B}\label{sec:thm2b}

Applying $2M_2 - 1$ annihilating polynomials $P_t(T_1)$ sequentially to the pullback $W$ grows an
ambient chain of $2M_2$ subspaces inside a $b M_2 k d$-dimensional ambient space, forcing the error
rank $r$ to scale linearly in $b k d$:

\begin{theorem}[Two-stage sumset recurrence rank bound]\label{thm:sumset-rank}
\formal{SumsetRankDeficitProofs.lean}{sumsetTwoHopRankBound}
\formal{SumsetRankDeficit.lean}{SumsetTwoHopRankBoundSpec}
Let $F$ be a field with $(d : F) \ne 0$, let $M_1, M_2, b \ge 1$, and let $1 \le k \le d$. Suppose
$p_1 : \Z_d \to \mathrm{Fin}\,M_1$ admits a shifted $(2M_2)$-fold monochromatic sumset
$(\widetilde{J}_0, \widetilde{B}, \mathrm{diamB})$ in class $m_1$ with
$\sum_{t=0}^{2M_2 - 2} \mathrm{diamB}_t \le d$. Then for every $p_2 : \Z_d \to \mathrm{Fin}\,M_2$
and every pair of $F$-linear class-uniform maps $K_1, K_2$ on $\Mat_{d\times d}(F^b)$, writing
$r := \rank_F(\tau - K_2 \circ K_1)$,
\[
  2r \;\ge\; b k \Bigl(d - \sum_{t=0}^{2M_2 - 2} \mathrm{diamB}_t\Bigr),
  \qquad\text{and if } \mathrm{diamB}_t \le \lfloor d/(4M_2)\rfloor \text{ for all } t,\quad
  4r \;\ge\; b k d.
\]
\end{theorem}
\begin{outline}
Since $(d : F) \ne 0$, $X^d - 1$ is separable over $F$; passing to its splitting field $E/F$
preserves $\rank(\tau - K_2 \circ K_1)$ and supplies an injective character $\chi : \Z_d \to E^\times$.
Let $W \le A_{\mathrm{band}}$ ($\dim_E A_{\mathrm{band}} \le b M_2 k d$) be the pullback from
\Cref{lem:band-pullback}(b) on $\widetilde{J}_0$, with $\dim_E W \ge b k d - r$. For each
$0 \le t < 2M_2 - 1$, let $P_t$ be the sparse annihilating polynomial on $\widetilde{B}_t$ from
\Cref{lem:sumset-poly}. Because $\chi(\nu \cdot -)$ is annihilated by $P_t$ while every shift in
$\widetilde{J}_0 + \sum_{s \le t} \widetilde{B}_s$ stays in $p_1^{-1}(m_1)$ where $K_1$ agrees with
$\tau \pm \text{error}$ on the error-free subspace, $P_t(T_1)$ maps the $t$-th span $U_t :=
\sum_{S \subseteq \{0,\dots,t-1\}} \bigl(\prod_{s\in S} T_{\widetilde{B}_s}\bigr) W$ into the kernel
of stacked $\widetilde{J}_0$-evaluation, which intersects $W$ trivially! Hence each of the
$2M_2 - 1$ steps adds at least $\dim_E W - b M_2 k\,\mathrm{diamB}_t$ dimensions inside
$A_{\mathrm{band}}$:
\[
  2M_2 \dim_E W - b M_2 k \sum_{t=0}^{2M_2 - 2} \mathrm{diamB}_t \;\le\; \dim_E A_{\mathrm{band}}
  \;\le\; b M_2 k d.
\]
Substituting $\dim_E W \ge b k d - r$ gives
$2 M_2(b k d - r) \le b M_2 k\bigl(d + \sum_t \mathrm{diamB}_t\bigr)$, i.e.,
$2r \ge bk\bigl(d - \sum_t \mathrm{diamB}_t\bigr)$. When every $\mathrm{diamB}_t \le \lfloor d/(4M_2)\rfloor$,
$\sum_t \mathrm{diamB}_t \le (2M_2 - 1)\frac{d}{4M_2} \le \frac{d}{2}$, giving $4r \ge b k d$.
\end{outline}

\begin{lemma}[Iterated K\H{o}v\'ari--S\'os--Tur\'an monochromatic sumset extraction]\label{lem:kst-sumset}
\formal{SumsetRankDeficitProofs.lean}{iteratedKovariSosTuranSumset}
\formal{SumsetRankDeficit.lean}{IteratedKovariSosTuranSumsetSpec}
For every $M_1, M_2 \ge 1$, there exist $0 < c_{\mathrm{lo}} \le C_{\mathrm{hi}}$ and $D_0 \in \N$
such that for all $d \ge D_0$, every $M_1$-coloring $p_1 : \Z_d \to \mathrm{Fin}\,M_1$ contains an
unshifted (and hence shifted) $(2M_2)$-fold monochromatic sumset in $[0, \lfloor d/(4M_2)\rfloor]$
at width scale $1 \le k \le d$ satisfying
\[
  c_{\mathrm{lo}}\,(\log d)^{\frac{1}{2M_2 - 1}} \;\le\; k \;\le\;
  C_{\mathrm{hi}}\,(\log d)^{\frac{1}{2M_2 - 1}}.
\]
\end{lemma}
\begin{outline}
Choose the densest color class $m_1$ inside $[0, \lfloor d/2\rfloor)$ and build a bipartite
addition table between blocks of length $\le \lfloor d/(4M_2)\rfloor$ and base intervals. At each of
the $2M_2 - 1$ stages, K\H{o}v\'ari--S\'os--Tur\'an \cite{KovariSosTuran1954} (formalized via
falling-factorial double counting of $(k+1)$-tuples of common neighbors) extracts $k_{t+1}$ left
vertices and $k + 1$ right shifts at exponential-in-$k^{2M_2-1}$ density cost, balancing at
$k = \Theta_{M_1,M_2}\bigl((\log d)^{1/(2M_2-1)}\bigr)$.
\end{outline}

\begin{theorem}[Theorem 2B: log-power two-stage rank deficit over admissible fields]\label{thm:2b}
\formal{SumsetRankDeficitProofs.lean}{theorem2B_TwoStageLogPowerRankDeficit}
\formal{SumsetRankConsequences.lean}{theorem2B_ExplicitConstant,theorem2BExplicitConst_pos}
\formal{SumsetRankDeficit.lean}{Theorem2B_TwoStageLogPowerRankDeficitSpec}
For all integers $M_1, M_2 \ge 1$, define the explicit positive constant and threshold
(\lean{theorem2BExplicitConst}, \lean{theorem2BExplicitThreshold})
\begin{align*}
  c(M_1, M_2) &\;:=\; \frac{1}{144\,\bigl(\log(16 M_1 M_2)\bigr)^{\frac{1}{2M_2 - 1}}} \;>\; 0, \\
  D_0(M_1, M_2) &\;:=\; \max\!\Bigl(4M_2,\; \bigl\lceil \exp\bigl(36^{2M_2-1}\log(16 M_1 M_2)\bigr)\bigr\rceil + 2\Bigr).
\end{align*}
Then for every field $F$, every $d \ge D_0(M_1,M_2)$ with $(d : F) \ne 0$, every lane width
$b \ge 1$, and every pair of $F$-linear class-uniform stages $K_1, K_2$ on $\Mat_{d\times d}(F^b)$
with $M_1$ and $M_2$ classes,
\[
  \rank_F(\tau - K_2 \circ K_1)
  \;\ge\;
  c(M_1, M_2)\,b\,d\,(\log d)^{\frac{1}{2M_2 - 1}}.
\]
\end{theorem}
\begin{outline}
Combine \Cref{lem:kst-sumset} (evaluated with explicit KST parameters at
$k = \bigl\lfloor \bigl(\frac{\log d}{18^{2M_2-1}\log(16M_1 M_2)}\bigr)^{1/(2M_2-1)}\bigr\rfloor$)
and \Cref{thm:sumset-rank} ($r \ge b k d / 4$).
\end{outline}

\begin{corollary}[Single-class second stage: $\Omega(b\,d\log d)$ rank deficit]\label{cor:2b-m2-one}
\formal{SumsetRankConsequences.lean}{theorem2B_OneSecondStage_OmegaLog}
When $M_2 = 1$ ($2M_2 - 1 = 1$), for every $M_1 \ge 1$ there exist $c > 0$ and $D_0 \ge 2$ such
that for every field $F$, every $d \ge D_0$ with $(d : F) \ne 0$, every $b \ge 1$, and every
$M_1$-class $K_1$ and $1$-class $K_2$ on $\Mat_{d\times d}(F^b)$,
\[
  \rank_F(\tau - K_2 \circ K_1) \;\ge\; c\,b\,d\log d.
\]
\end{corollary}
\begin{outline}
Specialize \Cref{thm:2b} to $M_2 = 1$, where $\frac{1}{2M_2 - 1} = 1$.
\end{outline}

\subsection{Single-class band relaxation obstruction at \texorpdfstring{$d = 12$}{d = 12} and the macroscopic target}\label{sec:d12-band}

Why did \Cref{thm:sumset-rank} use the \emph{actual} stage-1 class-uniform equality
$(K_2 K_1 X)_{:,c} = (\tau X)_{:,c}$ across shifted sumsets rather than merely relaxing stage~1 to
an abstract linear readout $L$ on $\leanname{SharedDecoders.ColumnAmbient}(H)$ for a single class
$J \subseteq \Z_d$ of density $1/2$ and a single $k = 2$ band? Because that abstract single-class
band relaxation is \emph{false}, even over $\C$ at $d = 12$!

\begin{example}[Unconditional $d = 12$ single-class band witness over $\C$]\label{ex:d12-band}
\formal{SumsetRankDeficitProofs.lean}{control_D12SingleClassAbstractBand}
\formal{SumsetRankConsequences.lean}{exists_primitive_twelfth_root_complex,d12AddChar_injective_and_primitive,control_D12_UnconditionalExists,control_D12_BandRowSpaceChannelEquiv}
\formal{SumsetRankDeficit.lean}{Control_D12SingleClassAbstractBandSpec}
At $d = 12$ and $k = 2$ with $q = 4$ channels (realized e.g.\ at $b M_2 = 2$ so $q = b M_2 k = 4$),
on the half-density subset $J_{12} = \{0, 1, 3, 4, 7, 9\} \subseteq \Z_{12}$, there exist
unconditionally a primitive $12$-th root of unity $\zeta_{12} = \exp(2\pi i / 12) \in \C$, a
shift-intertwining linear equivalence between $(\Z_{12} \to \C^4)$ and
$\leanname{SharedDecoders.ColumnAmbient}(\leanname{SharedDecoders.bandRowSpace}(\chi, p_0, 2))$
at lane width $b = 2$ and constant $1$-class partition $p_0(c) = 0$ ($M_2 = 1$), a linear readout
$L : (\Z_{12} \to \C^4) \to \C^{12}$, and a full $24$-dimensional subspace $W$ ($\dim_{\C} W = 24 =
kd$) on which stacked evaluation over $J_{12}$ is injective and realizes \emph{every} pair of band
vectors $(a_0, a_1) \in (\C^{12})^2$ as $L(T_c w) = a_0 + \zeta_{12}^c a_1$ for all $c \in J_{12}$.
\end{example}
\begin{outline}
Define $\zeta_{12} = \exp(2\pi i / 12) \in \C$, verify $\zeta_{12}^{12} = 1$ and $\zeta_{12}^m \ne 1$
for $1 \le m < 12$, and build the explicit $24$-dimensional $4$-channel encoding over $\Z_{12}$
whose $J_{12}$-shifts decode $a_0 + \zeta_{12}^c a_1$ exactly;
\lean{control_D12_BandRowSpaceChannelEquiv} identifies $(\Z_{12} \to \C^4)$ shift-equivariantly with
$\leanname{SharedDecoders.ColumnAmbient}$ of $\leanname{SharedDecoders.bandRowSpace}$ at
$b = 2, M_2 = 1, k = 2$.
\end{outline}

Consequently, pushing two-stage linear rank deficits from $\Omega\bigl(bd(\log d)^{1/(2M_2-1)}\bigr)$
all the way to $\eps(M_1, M_2) b d^2$ must couple \emph{multiple} stage-1 classes or \emph{multiple}
frequency bands simultaneously. We formalize both the general-field macroscopic target
(\lean{OpenTarget_MacroscopicTwoStageRankDeficitSpec}) and its $(d : F) \ne 0$ specialization
(\lean{OpenTarget_MacroscopicTwoStageRankDeficitAdmissibleFieldSpec}).

\begin{proposition}[Macroscopic two-stage rank target and low-error refutation]\label{prop:macroscopic-target}
\formal{SumsetRankDeficitProofs.lean}{openTarget_admissible_of_unrestricted}
\formal{MachineSweepConsequences.lean}{conditional_NoLowErrorTwoStageLinear_of_MacroscopicTarget}
\formal{SumsetRankDeficit.lean}{OpenTarget_MacroscopicTwoStageRankDeficitSpec,OpenTarget_MacroscopicTwoStageRankDeficitAdmissibleFieldSpec}
\lean{OpenTarget_MacroscopicTwoStageRankDeficitSpec} implies
\lean{OpenTarget_MacroscopicTwoStageRankDeficitAdmissibleFieldSpec}, and under
\lean{OpenTarget_MacroscopicTwoStageRankDeficitSpec}, for every fixed $M_1, M_2 \ge 1$ there exists
$\eps > 0$ such that along every sequence of fields $F_d$, lane widths $b_d \ge 1$, and
$M_1, M_2$-class linear stages $K_{1,d}, K_{2,d}$, the normalized rank error
$\rank_{F_d}(\tau - K_{2,d} \circ K_{1,d}) / (b_d d^2)$ cannot converge to $0$ as $d \to \infty$.
\end{proposition}
\begin{outline}
Immediate from the quantifier specialization and contrapositive of the rank lower bound.
\end{outline}
